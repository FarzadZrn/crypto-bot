def start_dummy_server():
    """ایجاد یک وب‌سرور مجازی برای زنده نگه داشتن بات روی هاست رایگان رندر"""
    handler = http.server.SimpleHTTPRequestHandler
    # رندر پورت را به صورت خودکار از ما می‌خواهد، در غیر این صورت روی پورت ۸۰۸۰ می‌رود
    port = int(os.environ.get("PORT", 8080))
    with socketserver.TCPServer(("", port), handler) as httpd:
        print(f"Dummy web server running on port {port}")
        httpd.serve_forever()

def get_crypto_data():
    try:
        headers = {'User-Agent': 'Mozilla/5.0'}
        # آدرس اصلاح شده و دقیق API عمومی کوین‌پاپریقا برای دریافت دامیننس و حجم زنده
        url = "https://coinpaprika.com"
        response = requests.get(url, headers=headers, timeout=15).json()
        
        btc_dominance = response.get("bitcoin_dominance_percentage", 59.3)
        volume_24h = response.get("volume_24h_usd", 0)
        volume_change = response.get("volume_24h_change_24h", 0)
        
        whale_alert = "💡 خنثی (نقدینگی در حال نوسان طبیعی بازار)"
        if volume_change > 15.0 and volume_24h > 50000000000:
            whale_alert = "🚨 ورود سنگین (نهنگ‌ها در حال پمپاژ پول هوشمند!)"
        elif volume_change < -15.0:
            whale_alert = "⚠️ خروج سرمایه (نهنگ‌ها در حال نقد کردن)"

        if btc_dominance > 58:
            altcoin_index = 42
        elif btc_dominance < 52:
            altcoin_index = 76
        else:
            altcoin_index = 50

        top_narratives = ["AI (هوش مصنوعی)", "RWA (دارایی‌های واقعی)", "DePIN (زیرساخت فیزیکی)"]
        return round(btc_dominance, 2), int(altcoin_index), top_narratives, whale_alert
    except Exception as e:
        print(f"Data Fetch Error: {e}")
        return None, None, [], "Error"

def send_telegram_alert(message):
    # آدرس رسمی، قطعی و تصحیح‌شده تلگرام
    url = f"https://telegram.org{TELEGRAM_TOKEN}/sendMessage"
    payload = {
        "chat_id": CHAT_ID,
        "text": message,
        "parse_mode": "Markdown"
    }
    try:
        response = requests.post(url, json=payload, timeout=15)
        if response.status_code == 200:
            print("Alert sent to Telegram successfully.")
        else:
            print(f"Telegram API Error: {response.text}")
    except Exception as e:
        print(f"Connection Error: {e}")

def analyze_market():
    print("Checking market conditions & Smart Money...")
    btc_dom, alt_idx, narratives, whale_signal = get_crypto_data()

    if btc_dom is None or alt_idx is None:
        return

    if alt_idx >= 75:
        status_text = "🚀 **آلت‌سیزن (Altseason) رسماً آغاز شد!**"
    elif alt_idx <= 25:
        status_text = "👑 **بیت‌کوین سیزن (Bitcoin Season) آغاز شد!**"
    else:
        status_text = "⚖️ **بازار در وضعیت خنثی قرار دارد.**"

    narratives_text = "\n".join([f"  🔥 {name}" for name in narratives])

    report_message = (
        f"{status_text}\n\n"
        f"🐋 **ردپای نهنگ‌ها و پول هوشمند:**\n"
        f"🔹 وضعیت نقدینگی نهنگ‌ها: `{whale_signal}`\n"
        f"🎯 روایت‌های برتر: \n{narratives_text}\n\n"
        f"📊 **شاخص‌های عمومی:**\n"
        f"🔹 شاخص آلت‌سیزن: `{alt_idx}/100`\n"
        f"📉 دامیننس بیت‌کوین: `{btc_dom}%`\n\n"
        f"⏰ رادار اختصاصی پول هوشمند کریپتو"
    )
    send_telegram_alert(report_message)

if __name__ == "__main__":
    # اجرای سرور مجازی وب در یک شاخه فرعی برای زنده ماندن روی رندر

    # اجرای اولین گزارش بلافاصله بعد از روشن شدن
